GERADOR DE KANBAN 100% DINÂMICO (CORRIGIDO PARA A SUA PLANILHA) const SHEET_ID = '1oZarWql1SOswIYMoHQkbVGg5tqjjgKAkp4JYK7qvdj4'; async function carregarKanbanDoSheets() { const boardContainer = document.getElementById('kanban-board'); try { // Força a leitura do intervalo A1 até Z100 const url = `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?tqx=out:json&range=A1:Z100`; const response = await fetch(url); const text = await response.text(); const jsonData = JSON.parse(text.substring(47, text.length - 2)); const table = jsonData.table; const rows = table.rows; boardContainer.innerHTML = ''; if (!rows || rows.length === 0) { boardContainer.innerHTML = '

Planilha vazia.

'; return; } // 1. Extrai os Títulos das Colunas (Linha 1 da planilha) const colunasMap = []; const empenhoHeaderRow = rows[0]; // Primeira linha da planilha if (empenhoHeaderRow && empenhoHeaderRow.c) { empenhoHeaderRow.c.forEach((cell, index) => { // Tenta pegar o nome pelo rótulo do Google ou pelo valor da célula A1, B1, C1... let nomeColuna = ''; if (table.cols && table.cols[index] && table.cols[index].label) { nomeColuna = table.cols[index].label; } else if (cell && cell.v !== null) { nomeColuna = String(cell.v); } if (nomeColuna.trim() !== '') { colunasMap.push({ index: index, nome: nomeColuna.trim(), cardsHTML: '' }); } }); } if (colunasMap.length === 0) { boardContainer.innerHTML = '

Nenhum título encontrado na Linha 1 da planilha.

'; return; } // 2. Percorre as Linhas de Dados (a partir da linha 2, índice 1) for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (!row || !row.c) continue; row.c.forEach((cell, cellIndex) => { if (cell && cell.v !== null && String(cell.v).trim() !== '') { const colObj = colunasMap.find(c => c.index === cellIndex); if (colObj) { colObj.cardsHTML += `

${String(cell.v)}

`; } } }); } // 3. Desenha as colunas dinâmicas no Dashboard colunasMap.forEach(col => { const colDiv = document.createElement('div'); colDiv.className = 'kanban-column'; colDiv.innerHTML = `
${col.nome}
${col.cardsHTML ? col.cardsHTML : '

Vazio

'}
`; boardContainer.appendChild(colDiv); }); } catch (error) { console.error("Erro ao conectar com o Google Sheets:", error); boardContainer.innerHTML = '

Erro ao carregar dados da planilha.

'; } } window.addEventListener('DOMContentLoaded', carregarKanbanDoSheets);
